# Site-Wide Bilingual Support Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build static Chinese/English support across the whole personal site, with reviewable MiMo-generated translations for long-form content.

**Architecture:** Add explicit locale primitives, bilingual UI messages, locale-prefixed static routes, and a local translation script that writes committed cache files. The browser and GitHub Pages build read local files only; MiMo API keys stay in local environment variables.

**Tech Stack:** Next.js App Router static export, React 19, TypeScript, Tailwind CSS, Vitest, Node.js scripts using `fetch`, MiMo OpenAI-compatible API.

```ts
import { defaultLocale, isLocale, type Locale } from "./locales";

function ensureLeadingSlash(pathname: string): string {
  return pathname.startsWith("/") ? pathname : `/${pathname}`;
}

function ensureTrailingSlash(pathname: string): string {
  if (pathname.includes("#") || pathname.includes("?")) return pathname;
  return pathname.endsWith("/") ? pathname : `${pathname}/`;
}

export function getLocaleFromPathname(pathname: string): Locale {
  const [, maybeLocale] = ensureLeadingSlash(pathname).split("/");
  return maybeLocale && isLocale(maybeLocale) ? maybeLocale : defaultLocale;
}

export function stripLocalePrefix(pathname: string): string {
  const normalized = ensureTrailingSlash(ensureLeadingSlash(pathname));
  const parts = normalized.split("/");
  const maybeLocale = parts[1];
  if (!maybeLocale || !isLocale(maybeLocale)) return normalized;
  const stripped = `/${parts.slice(2).join("/")}`;
  return stripped === "/" ? "/" : ensureTrailingSlash(stripped);
}

export function addLocalePrefix(pathname: string, locale: Locale): string {
  const stripped = stripLocalePrefix(pathname);
  if (locale === defaultLocale) return stripped;
  return stripped === "/" ? `/${locale}/` : `/${locale}${stripped}`;
}

export function switchLocalePathname(pathname: string, nextLocale: Locale): string {
  return addLocalePrefix(stripLocalePrefix(pathname), nextLocale);
}
```

```ts
import type { Locale } from "@/i18n/locales";

export interface LanguageDetectionResult {
  language: Locale | null;
  confidence: "high" | "ambiguous";
}

export function detectSourceLanguage(text: string): LanguageDetectionResult {
  const chineseMatches = text.match(/[\u3400-\u9fff]/g) ?? [];
  const englishMatches = text.match(/[A-Za-z]/g) ?? [];
  const chineseCount = chineseMatches.length;
  const englishCount = englishMatches.length;
  const totalSignal = chineseCount + englishCount;

  if (totalSignal < 20) {
    return { language: null, confidence: "ambiguous" };
  }

  const chineseRatio = chineseCount / totalSignal;
  const englishRatio = englishCount / totalSignal;

  if (chineseRatio >= 0.35 && chineseCount >= 8) {
    return { language: "zh", confidence: "high" };
  }

  if (englishRatio >= 0.75 && englishCount >= 20) {
    return { language: "en", confidence: "high" };
  }

  return { language: null, confidence: "ambiguous" };
}

export function getTargetLanguage(sourceLanguage: Locale): Locale {
  return sourceLanguage === "zh" ? "en" : "zh";
}
```

```ts
import { createHash } from "node:crypto";
import type { Locale } from "@/i18n/locales";

export interface TranslationCacheSummary {
  sourceHash?: string;
  targetLanguage?: Locale;
  body?: string;
}

export interface TranslationCacheExpectation {
  sourceHash: string;
  targetLanguage: Locale;
}

export function createSourceHash(source: string): string {
  return createHash("sha256").update(source).digest("hex");
}

export function isTranslationCacheFresh(
  cache: TranslationCacheSummary | null | undefined,
  expected: TranslationCacheExpectation
): boolean {
  return Boolean(
    cache?.sourceHash === expected.sourceHash &&
      cache.targetLanguage === expected.targetLanguage &&
      cache.body
  );
}
```

Create `scripts/translate-content.mjs` with these exported helpers and CLI behavior:

```js
#!/usr/bin/env node
import { createHash } from "node:crypto";
import { existsSync, mkdirSync, readFileSync, readdirSync, writeFileSync } from "node:fs";
import { basename, dirname, join } from "node:path";

const POSTS_DIR = "content/posts";
const CACHE_DIR = "content/generated/translations/posts";

export function parseFrontmatter(source) {
  if (!source.startsWith("---")) return { frontmatter: {}, body: source };
  const end = source.indexOf("\n---", 3);
  if (end === -1) return { frontmatter: {}, body: source };
  const raw = source.slice(3, end).trim();
  const body = source.slice(end + 4).trimStart();
  const frontmatter = {};
  let currentArrayKey = null;

  for (const line of raw.split("\n")) {
    const arrayItem = line.match(/^\s*-\s+(.+)$/);
    if (arrayItem && currentArrayKey) {
      frontmatter[currentArrayKey].push(arrayItem[1].replace(/^['"]|['"]$/g, ""));
      continue;
    }

    const match = line.match(/^([A-Za-z0-9_-]+):\s*(.*)$/);
    if (!match) continue;
    const [, key, value] = match;
    if (value === "") {
      frontmatter[key] = [];
      currentArrayKey = key;
    } else {
      frontmatter[key] = value.replace(/^['"]|['"]$/g, "");
      currentArrayKey = null;
    }
  }

  return { frontmatter, body };
}

export function createSourceHash(source) {
  return createHash("sha256").update(source).digest("hex");
}

export function toCachePath(postPath) {
  return join(CACHE_DIR, basename(postPath).replace(/\.mdx$/, ".json"));
}

export function detectLanguage(text, explicitLanguage) {
  if (explicitLanguage === "en" || explicitLanguage === "zh") return explicitLanguage;
  const chineseCount = (text.match(/[\u3400-\u9fff]/g) ?? []).length;
  const englishCount = (text.match(/[A-Za-z]/g) ?? []).length;
  const total = chineseCount + englishCount;
  if (total < 20) return null;
  if (chineseCount / total >= 0.35 && chineseCount >= 8) return "zh";
  if (englishCount / total >= 0.75 && englishCount >= 20) return "en";
  return null;
}

export function getTargetLanguage(sourceLanguage) {
  return sourceLanguage === "zh" ? "en" : "zh";
}

export function buildMiMoRequest({ model, sourceLanguage, targetLanguage, title, excerpt, tags, body }) {
  return {
    model,
    messages: [
      {
        role: "system",
        content:
          "You are a professional translation editor. Follow baoyu-translate normal mode: analyze first, then rewrite naturally in the target language. Preserve facts, logic, markdown, links, code blocks, and frontmatter semantics. Use a business/formal style. Return strict JSON only.",
      },
      {
        role: "user",
        content: JSON.stringify({
          sourceLanguage,
          targetLanguage,
          style: "business/formal",
          outputSchema: {
            sourceLanguage: "en|zh",
            targetLanguage: "en|zh",
            title: "string",
            excerpt: "string",
            tags: ["string"],
            body: "markdown string",
          },
          source: { title, excerpt, tags, body },
        }),
      },
    ],
    max_completion_tokens: 8192,
    temperature: 0.3,
    top_p: 0.95,
    stream: false,
  };
}

function readJsonIfExists(path) {
  if (!existsSync(path)) return null;
  return JSON.parse(readFileSync(path, "utf8"));
}

function listPostFiles() {
  if (!existsSync(POSTS_DIR)) return [];
  return readdirSync(POSTS_DIR)
    .filter((name) => name.endsWith(".mdx"))
    .map((name) => join(POSTS_DIR, name));
}

async function translateWithMiMo(request) {
  const apiKey = process.env.MIMO_API_KEY;
  const baseUrl = process.env.MIMO_BASE_URL ?? "https://token-plan-sgp.xiaomimimo.com/v1";
  if (!apiKey) throw new Error("MIMO_API_KEY is required when translation is needed.");

  const response = await fetch(`${baseUrl.replace(/\/$/, "")}/chat/completions`, {
    method: "POST",
    headers: {
      "api-key": apiKey,
      "Content-Type": "application/json",
    },
    body: JSON.stringify(request),
  });

  if (!response.ok) {
    throw new Error(`MiMo request failed: ${response.status} ${response.statusText}`);
  }

  const json = await response.json();
  const content = json.choices?.[0]?.message?.content;
  if (!content) throw new Error("MiMo response did not include message content.");

  try {
    return JSON.parse(content);
  } catch {
    mkdirSync("local/translation-diagnostics", { recursive: true });
    writeFileSync("local/translation-diagnostics/last-invalid-response.txt", content);
    throw new Error("MiMo returned invalid JSON. Raw response saved under local/translation-diagnostics/.");
  }
}

async function main() {
  mkdirSync(CACHE_DIR, { recursive: true });
  const model = process.env.MIMO_MODEL ?? "mimo-v2.5-pro";

  for (const postPath of listPostFiles()) {
    const source = readFileSync(postPath, "utf8");
    const sourceHash = createSourceHash(source);
    const { frontmatter, body } = parseFrontmatter(source);
    const sourceLanguage = detectLanguage(`${frontmatter.title ?? ""}\n${body}`, frontmatter.language);
    if (!sourceLanguage) {
      throw new Error(`${postPath}: language is ambiguous. Add language: en or language: zh to frontmatter.`);
    }

    const targetLanguage = getTargetLanguage(sourceLanguage);
    const cachePath = toCachePath(postPath);
    const existing = readJsonIfExists(cachePath);
    if (
      existing?.sourceHash === sourceHash &&
      existing?.targetLanguage === targetLanguage &&
      existing?.body
    ) {
      console.log(`skip ${postPath}: translation cache is fresh`);
      continue;
    }

    const translated = await translateWithMiMo(
      buildMiMoRequest({
        model,
        sourceLanguage,
        targetLanguage,
        title: frontmatter.title ?? "",
        excerpt: frontmatter.excerpt ?? "",
        tags: Array.isArray(frontmatter.tags) ? frontmatter.tags : [],
        body,
      })
    );

    const cache = {
      sourcePath: postPath,
      sourceHash,
      sourceLanguage,
      targetLanguage,
      model,
      mode: "normal",
      style: ["business", "formal"],
      ...translated,
    };

    mkdirSync(dirname(cachePath), { recursive: true });
    writeFileSync(cachePath, `${JSON.stringify(cache, null, 2)}\n`);
    console.log(`translated ${postPath} -> ${cachePath}`);
  }
}

if (import.meta.url === `file://${process.argv[1]}`) {
  main().catch((error) => {
    console.error(error.message);
    process.exit(1);
  });
}
```

Create `src/lib/content/posts.ts`:

```ts
import { existsSync, readFileSync, readdirSync } from "node:fs";
import { basename, join } from "node:path";
import type { Locale } from "@/i18n/locales";
import { defaultLocale } from "@/i18n/locales";
import { createSourceHash } from "./cache";
import { detectSourceLanguage, getTargetLanguage } from "./language";

const POSTS_DIR = "content/posts";
const CACHE_DIR = "content/generated/translations/posts";

export interface LocalizedPost {
  slug: string;
  title: string;
  excerpt: string;
  date?: string;
  tags: string[];
  body: string;
  locale: Locale;
}

function parseFrontmatter(source: string): { frontmatter: Record<string, unknown>; body: string } {
  if (!source.startsWith("---")) return { frontmatter: {}, body: source };
  const end = source.indexOf("\n---", 3);
  if (end === -1) return { frontmatter: {}, body: source };
  const raw = source.slice(3, end).trim();
  const body = source.slice(end + 4).trimStart();
  const frontmatter: Record<string, unknown> = {};
  let currentArrayKey: string | null = null;

  for (const line of raw.split("\n")) {
    const arrayItem = line.match(/^\s*-\s+(.+)$/);
    if (arrayItem && currentArrayKey) {
      (frontmatter[currentArrayKey] as string[]).push(arrayItem[1].replace(/^['"]|['"]$/g, ""));
      continue;
    }

    const match = line.match(/^([A-Za-z0-9_-]+):\s*(.*)$/);
    if (!match) continue;
    const [, key, value] = match;
    if (value === "") {
      frontmatter[key] = [];
      currentArrayKey = key;
    } else {
      frontmatter[key] = value.replace(/^['"]|['"]$/g, "");
      currentArrayKey = null;
    }
  }

  return { frontmatter, body };
}

export function getPostSlugs(): string[] {
  if (!existsSync(POSTS_DIR)) return [];
  return readdirSync(POSTS_DIR)
    .filter((name) => name.endsWith(".mdx"))
    .map((name) => basename(name, ".mdx"));
}

export function getLocalizedPost(slug: string, locale: Locale = defaultLocale): LocalizedPost {
  const postPath = join(POSTS_DIR, `${slug}.mdx`);
  const source = readFileSync(postPath, "utf8");
  const sourceHash = createSourceHash(source);
  const { frontmatter, body } = parseFrontmatter(source);
  const detected = detectSourceLanguage(`${String(frontmatter.title ?? "")}\n${body}`);
  const sourceLanguage = (frontmatter.language as Locale | undefined) ?? detected.language;
  if (!sourceLanguage) throw new Error(`${postPath}: ambiguous language.`);

  if (locale === sourceLanguage) {
    return {
      slug,
      title: String(frontmatter.title ?? slug),
      excerpt: String(frontmatter.excerpt ?? ""),
      date: typeof frontmatter.date === "string" ? frontmatter.date : undefined,
      tags: Array.isArray(frontmatter.tags) ? (frontmatter.tags as string[]) : [],
      body,
      locale,
    };
  }

  const targetLanguage = getTargetLanguage(sourceLanguage);
  if (locale !== targetLanguage) throw new Error(`${postPath}: unsupported locale ${locale}.`);

  const cachePath = join(CACHE_DIR, `${slug}.json`);
  if (!existsSync(cachePath)) throw new Error(`${postPath}: missing translation cache ${cachePath}.`);
  const cache = JSON.parse(readFileSync(cachePath, "utf8"));
  if (cache.sourceHash !== sourceHash || cache.targetLanguage !== targetLanguage) {
    throw new Error(`${postPath}: stale translation cache ${cachePath}. Run npm run translate:content.`);
  }

  return {
    slug,
    title: cache.title,
    excerpt: cache.excerpt ?? "",
    date: typeof frontmatter.date === "string" ? frontmatter.date : undefined,
    tags: cache.tags ?? [],
    body: cache.body,
    locale,
  };
}
```
